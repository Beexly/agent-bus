class CrossLayerVetoEngine:

        self.biomechanics_vetos = set()
        self.weather_vetos = set()

    def add_veto(self, layer: str, veto_id: str, condition: callable):
        """Register a veto constraint with associated verification function."""
        if layer == "physics":
            self.physics_vetos.add((veto_id, condition))
        elif layer == "biomechanics":
            self.biomechanics_vetos.add((veto_id, condition))
        elif layer == "weather":
            self.weather_vetos.add((veto_id, condition))

    def evaluate_all(self, input_data: dict) -> bool:
        """Evaluate all registered vetoes across layers."""
        physics_ok = all(cond(input_data) for _, cond in self.physics_vetos)
